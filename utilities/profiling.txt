# nsys profiling 
# add this line to get cpu access
$ sudo sh -c 'echo 1 > /proc/sys/kernel/perf_event_paranoid'

$ nsys profile \
  --trace=cuda,nvtx,osrt \
  --cuda-memory-usage=true \
  --sample=cpu \
  --output=rl_profile \
  --force-overwrite=true \
  --stats=true \
  python rl_pfl.py
